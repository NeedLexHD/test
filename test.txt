/**
 * Helper function to retrieve GitHub images with caching
 */
function get_product_github_images($product_id) {
    static $cache = [];
    if (isset($cache[$product_id])) {
        return $cache[$product_id];
    }
    
    $images = get_post_meta($product_id, '_github_images', true);
    $cache[$product_id] = (!empty($images) && is_array($images)) ? $images : [];
    
    return $cache[$product_id];
}


// 1. PAGES PRODUITS - Afficher les images GitHub sur les pages produits
add_filter('woocommerce_product_get_image', 'display_github_product_image', 10, 5);
function display_github_product_image($image, $product, $size, $attr, $placeholder) {
    $product_id = $product->get_id();
    $github_images = get_product_github_images($product_id);
    if (!empty($github_images)) {
        $first_image = $github_images[0];
        $default_attr = array(
            'src' => $first_image,
            'class' => 'attachment-woocommerce_thumbnail size-woocommerce_thumbnail',
            'alt' => $product->get_name(),
        );
        $attr = wp_parse_args($attr, $default_attr);


        $html_attr = '';
        foreach ($attr as $name => $value) {
            $html_attr .= " $name=" . '"' . esc_attr($value) . '"';
        }


        return '<img' . $html_attr . ' />';
    }


    return $image;
}


// 2. PANIER - Afficher les images GitHub dans le panier
add_filter('woocommerce_cart_item_thumbnail', 'display_github_cart_thumbnail', 10, 3);
function display_github_cart_thumbnail($image, $cart_item, $cart_item_key) {
    $product = $cart_item['data'];
    $product_id = $product->get_id();
    $github_images = get_product_github_images($product_id);
    if (!empty($github_images)) {
        $first_image = $github_images[0];
        return '<img src="' . esc_url($first_image) . '" alt="' . esc_attr($product->get_name()) . '" class="attachment-woocommerce_thumbnail" />';
    }


    return $image;
}


// 3. RÉSUMÉ DE COMMANDE (CHECKOUT) - Afficher les images GitHub
add_filter('woocommerce_order_item_thumbnail', 'display_github_checkout_thumbnail', 10, 2);
function display_github_checkout_thumbnail($image, $item) {
    $product = $item->get_product();
    if (!$product) {
        return $image;
    }


    $product_id = $product->get_id();
    $github_images = get_product_github_images($product_id);


    if (!empty($github_images)) {
        $first_image = $github_images[0];
        return '<img src="' . esc_url($first_image) . '" alt="' . esc_attr($product->get_name()) . '" class="attachment-woocommerce_thumbnail" width="60" />';
    }


    return $image;
}


// 4. AJAX - Récupérer les images GitHub (support batch et single)
add_action('wp_ajax_get_github_image', 'ajax_get_github_image');
add_action('wp_ajax_nopriv_get_github_image', 'ajax_get_github_image');
function ajax_get_github_image() {
    if (isset($_POST['product_slugs']) && is_array($_POST['product_slugs'])) {
        $slugs = array_map('sanitize_text_field', $_POST['product_slugs']);
        $results = array();
        
        foreach ($slugs as $slug) {
            $product = get_page_by_path($slug, OBJECT, 'product');
            if ($product) {
                $github_images = get_product_github_images($product->ID);
                if (!empty($github_images)) {
                    $results[$slug] = $github_images[0];
                }
            }
        }
        
        wp_send_json_success($results);
    }
    
    $slug = isset($_POST['product_slug']) ? sanitize_text_field($_POST['product_slug']) : '';
    if (empty($slug)) {
        wp_send_json_error();
    }


    $product = get_page_by_path($slug, OBJECT, 'product');
    if (!$product) {
        wp_send_json_error();
    }


    $github_images = get_product_github_images($product->ID);
    if (!empty($github_images)) {
        wp_send_json_success($github_images[0]);
    }


    wp_send_json_error();
}


// 5. CHECKOUT BLOCKS - Hook l'API REST pour forcer les images GitHub
add_filter('woocommerce_rest_prepare_shop_order_object', 'force_github_in_rest_api', 10, 3);
function force_github_in_rest_api($response, $object, $request) {
    if (isset($response->data['line_items'])) {
        foreach ($response->data['line_items'] as &$item) {
            if (isset($item['product_id'])) {
                $github_images = get_product_github_images($item['product_id']);
                if (!empty($github_images)) {
                    $item['image'] = array(
                        'id' => 0,
                        'src' => $github_images[0]
                    );
                }
            }
        }
    }
    
    return $response;
}

add_action('wp_footer', 'script_variation_couleur_images', 20);
function script_variation_couleur_images() {

        return;
    }
    
    global $product;
    if (!$product || !$product->is_type('variable')) {
        return;
    }

        return;
    }

    $folder = ($prefix === 'LP') ? 'lampes/' : 'porte-cles/';
    
    $slug_parts = explode('-', $product_slug);
    $generic_words = array('porte', 'cle', 'cles', 'lampe', 'led', 'lumiere');
    $clean_parts = array();
    
    foreach ($slug_parts as $part) {
        if (!in_array(strtolower($part), $generic_words) && strlen($part) > 1) {
            $clean_parts[] = $part;
        }
    }
    
    if (!empty($clean_parts)) {
        $short_slug = implode('-', $clean_parts);
    } else {
        $short_slug = end($slug_parts);

                prefix: '<?php echo esc_js($prefix); ?>',
                folder: '<?php echo esc_js($folder); ?>',
                productSlug: '<?php echo esc_js($short_slug); ?>',
                debug: false
            },
            
            couleurMapping: {
                'arc-en-ciel': 'arc-en-ciel','blanc': 'blanc','bleu-metal': 'bleu-metal',
                'bleu-phosphorescent': 'bleu-vert-phosphorescent','gris': 'gris','magenta': 'magenta',
                'noir-et-blanc': 'noir-blanc','rose': 'rose','rouge-et-noir': 'rouge-noir',
                'transparent-luciole-bleu': 'luciole','tricolore-bleu-vert-violet': 'bleu-vert-violet',
                'tricolore-rouge-jaune-vert': 'rouge-jaune-vert','turquoise': 'turquoise','vert': 'vert',
                'vert-menthe': 'vert-menthe','vert-militaire': 'vert-militaire','vert-nucleaire': 'vert-nucleaire',
                'violet-brillant': 'violet-brillant'
            },
            
            typeMapping: {'anneau': 'anneau','crochet': 'crochet'},
            
            modeleMapping: {
                'grand-sourire': 'grand','sourire-leger': 'leger','sourire-malefique': 'malefique',
                'sourire-nauseeux': 'nauseuse','sourire-cousu': 'cousue'
            },
            
            log: function(message, data) {
                if (this.config.debug) console.log('[Variation↔Image] ' + message, data || '');
            },
            
            normalizeSlug: function(value) {
                return (value || '').toLowerCase()
                    .replace(/[éèêë]/g, 'e').replace(/[àâä]/g, 'a').replace(/[îï]/g, 'i')
                    .replace(/[ôö]/g, 'o').replace(/[ùûü]/g, 'u').replace(/[ç]/g, 'c')
                    .replace(/[\s_\(\)]/g, '-').replace(/[^a-z0-9\-]/g, '').replace(/\-+/g, '-').replace(/^-|-$/g, '');
            },
            
            getAllSelectedAttributes: function() {
                const attrs = {};
                $('select[name^="attribute_"]').each((i, el) => {
                    const name = $(el).attr('name');
                    const raw = $(el).val();
                    if (!raw) return;
                    const attrKey = name.replace(/^attribute_/, '');
                    attrs[attrKey] = this.normalizeSlug(raw);
                });
                return attrs;
            },
            
            init: function() {
                this.log('Init - ' + this.config.prefix + '-' + this.config.productSlug);
                this.bindEvents();
                this.syncThumbnailsWithAttributes();
                this.selectDefaultAttributes();
            },
            
            bindEvents: function() {
                const self = this;
                $('form.variations_form').on('found_variation', function(event, variation) {
                    if (window.manualThumbnailClick) return;
                    self.onVariationChange(variation);
                });
                $('select[name^="attribute_"]').on('change', function() {
                    if (window.manualThumbnailClick) return;
                    self.changeImageFromCurrentAttributes();
                });
            },
            
            syncThumbnailsWithAttributes: function() {
                const self = this;
                setTimeout(() => {
                    $('.flex-control-thumbs li img').each(function() {
                        const $thumb = $(this);
                        const imgSrc = $thumb.attr('src');
                        const attrs = self.extractAttributesFromImageUrl(imgSrc);
                        if (attrs.type) $thumb.attr('data-type', attrs.type);
                        if (attrs.modele) $thumb.attr('data-modele', attrs.modele);
                        if (attrs.couleur) $thumb.attr('data-couleur', attrs.couleur);
                    });
                }, 1200);
            },
            
selectDefaultAttributes: function() {
    $('select[name^="attribute_"]').each((i, el) => {
        const $select = $(el);
        // Si une valeur est déjà sélectionnée par défaut par WooCommerce, on touche à rien
        if ($select.val()) return;
        
        // Sinon on prend la première option valide
        const first = $select.find('option').filter((_, opt) => opt.value !== '').first();
        if (first.length) {
            $select.val(first.val());
            $select.trigger('change'); // <--- C'est ça qui débloque le panier !
        }
    });
    // On laisse le trigger('change') faire le boulot pour l'image aussi
},
            
            extractAttributesFromImageUrl: function(imageUrl) {
                if (!imageUrl) return {};
                const filename = imageUrl.split('/').pop().replace('.png', '').replace(/-\d+$/, '');
                const pattern = this.config.prefix + '-' + this.config.productSlug + '-';
                const result = {};
                if (filename.includes(pattern)) {
                    let suffix = filename.replace(pattern, '');
                    if (suffix.includes('-demo-')) suffix = suffix.split('-demo-')[0];
                    
                    if (this.config.prefix === 'PC' && (suffix === 'anneau' || suffix === 'crochet')) {
                        result.type = suffix;
                        return result;
                    }
                    if (this.config.prefix === 'LP') {
                        for (const [key, value] of Object.entries(this.modeleMapping)) {
                            if (value === suffix) { result.modele = key; return result; }
                        }
                    }
                    for (const [key, value] of Object.entries(this.couleurMapping)) {
                        if (value === suffix || key === suffix) { result.couleur = key; return result; }
                    }
                    result.couleur = suffix;
                }
                return result;
            },
            
            onVariationChange: function(variation) {
                const attrs = {};
                Object.keys(variation.attributes).forEach(key => {
                    if (!variation.attributes[key]) return;
                    const cleanKey = key.replace(/^attribute_/, '');
                    attrs[cleanKey] = this.normalizeSlug(variation.attributes[key]);
                });
                this.changeImageFromAttributes(attrs);
            },
            
            buildImageUrl: function(attrs) {
                let suffix = null;
                
                if (this.config.prefix === 'PC') {
                    const type = attrs['pa__type'] || attrs['_type'] || attrs['pa_type'] || attrs['type'] || null;
                    if (type && this.typeMapping[type]) {
                        suffix = this.typeMapping[type];
                    } else {
                        const couleur = attrs['pa_couleur'] || attrs['couleur'] || null;
                        if (couleur) suffix = this.couleurMapping[couleur] || couleur;
                    }
                }
                
                if (this.config.prefix === 'LP') {
                    const modele = attrs['pa_modele'] || attrs['modele'] || null;
                    if (modele && this.modeleMapping[modele]) {
                        suffix = this.modeleMapping[modele];
                    } else {
                        const couleur = attrs['pa_couleur'] || attrs['couleur'] || null;
                        if (couleur) suffix = this.couleurMapping[couleur] || couleur;
                    }
                }
                
                return suffix 
                    ? this.config.baseUrl + this.config.folder + this.config.prefix + '-' + this.config.productSlug + '-' + suffix + '.png'
                    : this.config.baseUrl + this.config.folder + this.config.prefix + '-' + this.config.productSlug + '.png';
            },
            
            changeImageFromCurrentAttributes: function() {
                const attrs = this.getAllSelectedAttributes();
                this.changeImageFromAttributes(attrs);
            },
            
            changeImageFromAttributes: function(attrs) {
                const imageUrl = this.buildImageUrl(attrs);
                this.updateMainImage(imageUrl);
            },
            
            updateMainImage: function(imageUrl) {
                const mainImage = $('.woocommerce-product-gallery__image img, .wp-post-image').first();
                if (!mainImage.length) return;
                const self = this;
                const tempImg = new Image();
                tempImg.onload = () => {
                    mainImage.css('opacity', '0.7');
                    setTimeout(() => {
                        mainImage.attr('src', imageUrl).removeAttr('srcset');
                        mainImage.closest('a').attr('href', imageUrl);
                        mainImage.css('opacity', '1');
                        self.updateActiveThumbnail(imageUrl);
                    }, 30);
                };
                tempImg.onerror = () => {
                    const altUrl = imageUrl.replace('.png', '-1.png');
                    const tempImg2 = new Image();
                    tempImg2.onload = () => {
                        mainImage.css('opacity', '0.7');
                        setTimeout(() => {
                            mainImage.attr('src', altUrl).removeAttr('srcset');
                            mainImage.closest('a').attr('href', altUrl);
                            mainImage.css('opacity', '1');
                            self.updateActiveThumbnail(altUrl);
                        }, 30);
                    };
                    tempImg2.onerror = () => {};
                    tempImg2.src = altUrl;
                };
                tempImg.src = imageUrl;
            },
            
            updateActiveThumbnail: function(imageUrl) {
                $('.flex-control-thumbs li img').removeClass('active flex-active');
                $('.flex-control-thumbs li img').each(function() {
                    if ($(this).attr('src') === imageUrl) {
                        $(this).addClass('active flex-active');
                        const dataIndex = $(this).attr('data-gallery-index');
                        if (dataIndex !== undefined) window.galleryCurrentIndex = parseInt(dataIndex);
                        const slider = $('.flex-control-thumbs');
                        if (slider.length) {
                            const thumbLeft = $(this).parent().position().left + slider.scrollLeft();
                            const thumbWidth = $(this).width();
                            const sliderWidth = slider.width();
                            const targetScroll = thumbLeft - (sliderWidth / 2) + (thumbWidth / 2);
                            requestAnimationFrame(() => slider.stop(true, true).animate({scrollLeft: targetScroll}, 80));
                        }
                    }
                });
            }
        };

                    window.galleryCurrentIndex = 0;
